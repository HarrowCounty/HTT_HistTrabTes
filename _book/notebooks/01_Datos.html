Por ejemplo, ya aquí empieza a haber detalles que los registraremos uno por uno mas adelantes.

Baja California Sur NO tiene un municipio llamado “GUERRERO NEGRO”. Hay una localidad oficial llamada así. ¿Cuantas localidades llamadas o conocidas por “Guerrero Negro” (oficiales o extra-oficiales) puede haber en Baja California Sur?.

Podemos pensar que el que levantó la información se refería a la localidad “GUERRERO NEGRO” en el municipio MUELEGUE, ahora, ¿Es correcto forzar la asignación de esta incautación al municipio de MUELEGUE solo por que sospechamos que el que levantó la información puso el nombre de la localidad y no del municipio? Al tratarse de datos tan sensibles como la incautación de droga no nos atrevemos a asignar datos a municipios por errores de captura. 5.1.1 Graficos generales de incautación

total_promedio <- Datos_DEFENSA |> group_by(AÑO,DROGA,ESTADO) |> summarise(
  Total_Kg  = sum(KGS.,na.rm = TRUE), 
  Total_Lt =sum(LTS.,na.rm = TRUE), 
  Total_Amp = sum(AMPOLLETA,na.rm = TRUE), 
  Total_Uni = sum(UNIDADES,na.rm = TRUE), 
  Prom_Kg = mean(KGS.,na.rm=TRUE),
  Prom_Lt = mean(LTS.,na.rm=TRUE),
  Prom_Amp = mean(AMPOLLETA,na.rm=TRUE),
  Prom_Uni = mean(UNIDADES,na.rm=TRUE),
  n = n(),
  .groups = "drop")

gtp1<-ggplot(total_promedio, aes(x=AÑO, y= Total_Kg, fill=DROGA))+geom_col()+labs(y="Total de Kg de droga incautado",x="AÑO", title="Total de  KG de Droga Incautado por año")+
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

gtp2 <- ggplot(total_promedio,aes(x=AÑO,y=Total_Lt,fill=DROGA))+geom_col()+labs(y="Litros de Droga",x="Año",title="Litros de droga incautada por año Escala Pseudo Log10")+scale_y_continuous(trans = pseudo_log_trans(base = 10))+
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

gtp3 <- ggplot(total_promedio, aes(x=AÑO,y=Total_Amp, fill=DROGA))+geom_col()+labs(y="Ampolletas de Droga",x="Año",title="Ampolletas de droga incautada por año Escala Pseudo Log10")+scale_y_continuous(trans = pseudo_log_trans(base = 10))+
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

gtp4 <- ggplot(total_promedio,aes(x=AÑO, y=Total_Uni, fill = DROGA))+geom_col()+labs(y="Unidades de Droga",x="Año",title="Unidades de droga incautada por año Escala Pseudo Log10")+scale_y_continuous(trans = pseudo_log_trans(base = 10))+
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

5.1.3 Analisis de series de tiempo y tendencia

Los siguientes graficos son para ver como cambia la incautación de la droga a lo largo de los años o si hay patrones según el mes.

Para ser mas simples a partir de aquí la exploración de datos solo se hará en el formato Kgs y Lts, dejando fuera las incautaciones en ampolletas y unidades.

incautacion_anual <- Datos_DEFENSA |> group_by(AÑO,DROGA) |>  summarise(
  #Resumen por AME: Año
  n=n(),
  Kgs_A = sum(KGS., na.rm = TRUE),
  Lts_A = sum(LTS.,na.rm = TRUE),
  .groups = "drop"
)

IA1<-ggplot(incautacion_anual,aes(x=AÑO, y=Kgs_A, color=DROGA, group = DROGA))+geom_line(linewidth = 1)+geom_point()+
  theme(axis.text.x = element_text(angle = 70, hjust = 1))+labs(x = "Año", y = "Kilogramos asegurados",title = "Incautación anual de droga")
IA2<-ggplot(incautacion_anual,aes(x=AÑO, y=Kgs_A, color=DROGA, group = DROGA))+geom_line(linewidth = 1)+geom_point()+
  theme(axis.text.x = element_text(angle = 70, hjust = 1))+scale_y_continuous(trans = pseudo_log_trans(base = 10))+labs(x = "Año", y = "Kilogramos asegurados",title = "Incautación anual de droga Pseudo Log 10")

IA3<-ggplot(incautacion_anual,aes(x=AÑO, y=Lts_A, color=DROGA, group = DROGA))+geom_line(linewidth = 1)+geom_point()+
  theme(axis.text.x = element_text(angle = 70, hjust = 1))+labs(x = "Año", y = "Litros asegurados",title = "Incautación anual de droga")
IA4<-ggplot(incautacion_anual,aes(x=AÑO, y=Lts_A, color=DROGA, group = DROGA))+geom_line(linewidth = 1)+geom_point()+
  theme(axis.text.x = element_text(angle = 70, hjust = 1))+scale_y_continuous(trans = pseudo_log_trans(base = 10))+labs(x = "Año", y = "Litros asegurados",title = "Incautación anual de droga Pseudo Log 10")
