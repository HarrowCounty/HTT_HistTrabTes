total_promedio <- Datos_DEFENSA_ETP |> group_by(AÑO,DROGA,ESTADO) |> summarise(
  Total_Kg  = sum(KGS.,na.rm = TRUE), 
  Total_Lt =sum(LTS.,na.rm = TRUE), 
  Total_Amp = sum(AMPOLLETA,na.rm = TRUE), 
  Total_Uni = sum(UNIDADES,na.rm = TRUE), 
  Prom_Kg = mean(KGS.,na.rm=TRUE),
  Prom_Lt = mean(LTS.,na.rm=TRUE),
  Prom_Amp = mean(AMPOLLETA,na.rm=TRUE),
  Prom_Uni = mean(UNIDADES,na.rm=TRUE),
  n = n(),
  .groups = "drop")

gtp1<-ggplot(total_promedio, aes(x=AÑO, y= Total_Kg, fill=DROGA))+geom_col()+labs(y="Total de Kg de droga incautado",x="AÑO", title="Total de  KG de Droga Incautado por año")+

gtp2 <- ggplot(total_promedio,aes(x=AÑO,y=Total_Lt,fill=DROGA))+geom_col()+labs(y="Litros de Droga",x="Año",title="Litros de droga incautada por año Escala Pseudo Log10")+scale_y_continuous(trans = pseudo_log_trans(base = 10))+
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

gtp3 <- ggplot(total_promedio, aes(x=AÑO,y=Total_Amp, fill=DROGA))+geom_col()+labs(y="Ampolletas de Droga",x="Año",title="Ampolletas de droga incautada por año Escala Pseudo Log10")+scale_y_continuous(trans = pseudo_log_trans(base = 10))+
  theme(axis.text.x = element_text(angle = 45, hjust = 1))
