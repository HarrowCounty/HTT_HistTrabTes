Este capítulo es para hacer un analísis temporal en terminos de:

¿Qué partido político gobernaba en el estado y en el municipio en donde fué la incautación?

La idea es ver si existe una tendencia

Entonces haré 32 gráficas, una por estado.

Cada gráfica tiene un panel por municipio, es decir, el número de paneles depende de cuántos municipios de ese estado tienen registros de incautación de cocaína.

En cada panel, cada punto es un año con incautación registrada. La altura del punto es la cantidad de kilogramos incautados ese año, y su color es el partido político que gobernaba el municipio. El fondo de cada año es el partido político que gobernaba el estado. Los puntos no se unen con líneas, porque no se asume nada de los años intermedios: un año sin punto significa que no hay incautación registrada.

colores_punto <- c(

#FUNCIÓN QUE SIMPLIFICA EL PARTIDO 
limpia_partido <- function(x) {
  # Toma el texto antes del primer guion ("pvem-pt-morena" -> "pvem") y lo pone en mayúsculas
  primero <- str_to_upper(str_extract(x, "^[^-]+"))
  # Si ese partido está en el catálogo se queda, si no se llama "OTROS"
  base <- if_else(primero %in% c("MORENA","PAN","PRI","PRD","MC","PVEM","PT"), primero, "OTROS")
  # Si el texto original tenía guion es coalición: se agrega "COALICIÓN " al inicio
  if_else(str_detect(x, "-") & base != "OTROS", paste("COALICIÓN", base), base)
}

#PUNTOS: cocaína por municipio y año
puntos <- Datos_DEFENSA_ETP |>
  filter(DROGA == "COCAINA") |>                                # solo cocaína
  mutate(PartidoMun = limpia_partido(PartidoElectoMun)) |>     # partido municipal simplificado
  group_by(ESTADO, MUNICIPIO, AÑO) |>                          # un renglón por municipio y año
  summarise(
    Total_Kgs  = sum(KGS., na.rm = TRUE),                      # suma de kg de ese año
    PartidoMun = PartidoMun[which.max(KGS.)],                  # si hay 2 partidos en el año, el del registro con más kg
    .groups = "drop"
  )

#FONDO: partido estatal por estado y año 
# Se toma de TODOS los registros (cualquier droga y municipio) porque el
# gobierno estatal es el mismo para todo el estado en un año dado.
estatal <- Datos_DEFENSA_ETP |>
  mutate(PartidoEst = limpia_partido(PartidoElectoEst)) |>
  distinct(ESTADO, AÑO, PartidoEst)                            # un renglón por estado-año-partido

# Revisión: debe salir vacío. Si sale algo, hay dos partidos en un mismo estado-año.
estatal |> count(ESTADO, AÑO) |> filter(n > 1)
# A tibble: 0 × 3
# ℹ 3 variables: ESTADO <chr>, AÑO <dbl>, n <int>
# Repite el fondo del estado en cada uno de sus municipios
fondo <- puntos |>
  distinct(ESTADO, MUNICIPIO) |>
  inner_join(estatal, by = "ESTADO", relationship = "many-to-many")

#GRÁFICA POR ESTADO
plot_estado <- function(est) {
  ggplot() +
    # Fondo: un rectángulo por año, coloreado con el partido estatal (sin alpha, el color ya es pálido)
    geom_rect(data = filter(fondo, ESTADO == est),
              aes(xmin = AÑO - 0.5, xmax = AÑO + 0.5, ymin = -Inf, ymax = Inf, fill = PartidoEst)) +
    # Puntos: primero un punto negro más grande, que funciona como contorno (sin color en aes, no genera leyenda)
    geom_point(data = filter(puntos, ESTADO == est),
               aes(AÑO, Total_Kgs), color = "black", size = 3.6) +
    # Puntos: encima, un punto por año con incautación, coloreado con el partido municipal
    geom_point(data = filter(puntos, ESTADO == est),
               aes(AÑO, Total_Kgs, color = PartidoMun), size = 2.6) +
    # Un panel por municipio; con ncol = 4 caben en 4 columnas
    facet_wrap(~ MUNICIPIO, ncol = 4, scales = "free_y",
               labeller = label_wrap_gen(width = 20)) +
    scale_fill_manual(values = colores_fondo) +                # usa el catálogo pálido para el fondo
    scale_color_manual(values = colores_punto) +               # usa el catálogo fuerte para los puntos
    scale_x_continuous(breaks = seq(2005, 2025, by = 5)) +     # un número cada 5 años para que no se encimen
    scale_y_continuous(labels = scales::comma) +               # kg con separador de miles
    labs(title = paste("Cocaína asegurada -", est),
         subtitle = "Fondo = partido estatal | Punto = partido municipal | Sin punto = sin incautación",
         fill = "Partido estatal (fondo)", color = "Partido municipal (punto)",
         x = "Año", y = "Kg") +
    theme_minimal() +
    theme(legend.position = "bottom",
          strip.text = element_text(size = 8, face = "bold"),  
          panel.spacing = unit(1, "lines"))                    
}

#IMPRIME UNA GRÁFICA POR ESTADO 
walk(unique(puntos$ESTADO), ~ print(plot_estado(.x)))
